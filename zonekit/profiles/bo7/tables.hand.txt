# Hand tables for Black Ops 7 (zonekit profile bo7): readers the machine-code transcriber cannot
# render. Each cites the disassembly of the decrypted cod.exe (flattened: file offset == RVA).

# streaminginfo's 0x389f050 (an array of 16-byte {u16* counts, void** slots}, called with the struct
# holding the count u32 at 0x4c as its fifth argument): the counts (u32@%0x4c of them) at 0, then the
# slots at 8, each read with the count 0x3ad5a00 finds: the u16 at (slot - slots) / 8 in the counts
# (0x389f207). Each pointer through DB_RewindPrepare (0x382f020: bit 63 an 8-byte descriptor first;
# bit 60 or 63 the data follows). Modern Warfare 2019's 0x2e42f40 has the same shape.
### f389f050
SELF|0x10
IF|u64@0x0 != 0
IF|(u64@0x0 >> 63) != 0
READ|8
ENDIF
IF|(((u64@0x0 & 0x9000000000000000) != 0) & 0xff) != 0
MARK|bo7_si_counts
READ|(u32@%0x4c + u32@%0x4c)
ENDIF
ENDIF
IF|u64@0x8 != 0
IF|(u64@0x8 >> 63) != 0
READ|8
ENDIF
IF|(((u64@0x8 & 0x9000000000000000) != 0) & 0xff) != 0
ARRAY|u32@%0x4c|0x8|f389f050_e1
ENDIF
ENDIF

### f389f050_e1
SELF|0x8
IF|u64@0x0 != 0
IF|(u64@0x0 >> 63) != 0
READ|8
ENDIF
IF|(((u64@0x0 & 0x9000000000000000) != 0) & 0xff) != 0
READ|(u16[($bo7_si_counts + (# * 2))] * 2)
ENDIF
ENDIF

# streaminginfo's 0x38a4120, its lists of 16-byte {name*, u32 type hash} at 0x80 (u32@0xbc of them)

# poolInfo's byte 0x29 of the type (0xb3fdae0 + 0x29 + 48 * type, the static table 0x3a42ff0 reads):
# set for xmodel, attachment, weapon and xcompositemodel only, whose name is then read (0x38a4293,
# through DB_RewindPrepare). The walkers have no type map: the four FNV-1a 32 hashes.
### f38a4120_e1

IF|u64@0x0 != 0
IF|(u64@0x0 >> 63) != 0
READ|8
ENDIF
IF|(((u64@0x0 & 0x9000000000000000) != 0) & 0xff) != 0
STR+|0x0
ENDIF
ENDIF
ENDIF

# sightconfig 0x3890b20 (0x38): its 0xa0-byte elements (i32@0x30 of them at 0x28, through

# allocation (0x3890c6d), then by the byte at 0x30: 1 a rewind lookup inline (0x3890ca6, the hash

ENDIF

# navmesh's 0x38748a0, its 0x68-byte records (i32@0xc of them at 0x10): 4 bytes, then two blocks
# through DB_RewindPrepare: at 0x58 i32@0xc bytes, at 0x60 i32@0xc bytes when the u8 at 0x54 is
# set, else none (0x3874a65: cmp byte [rsi-0xc], 0 / mov ecx, [rsi-0x54] / xor ecx, ecx, rsi the
# record's 0x60). The transcriber lost that select (UNSUPPORTED LS); Paradox Junction's server
# zone (srv_zm_sat_mantis, 15/10/2026 update) is the first seen with 0x60 set.
### f38748a0_e1
SELF|0x68
READ|4
IF|u64@0x58 != 0
IF|(u64@0x58 >> 63) != 0
READ|8
ENDIF
READ|i32@0xc
ENDIF
IF|u64@0x60 != 0
IF|(u64@0x60 >> 63) != 0
READ|8
ENDIF
IF|u8@0x54 != 0
READ|i32@0xc
ENDIF
ENDIF
